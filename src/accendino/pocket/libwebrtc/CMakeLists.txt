#  Install project for a libwebrtc binary distribution (https://github.com/webrtc-sdk/libwebrtc/releases), used
#  by libwebrtc.accendino as cmakeProjectDir: the distribution has no build system, it only contains include/,
#  lib/ and LICENSE. LIBWEBRTC_DIST_DIR is the directory where it has been extracted, LIBWEBRTC_VERSION its
#  version (like m150.7871.04).
#
#  The distribution is installed in the deploy tree as:
#    * <includedir>/libwebrtc/...: the headers, they include each other as "rtc_types.h" or "base/refcount.h"
#      so consumers use -I<includedir>/libwebrtc
#    * <libdir>/libwebrtc.so on Linux, <bindir>/libwebrtc.dll and <libdir>/libwebrtc.dll.lib on Windows
#    * <libdir>/cmake/LibWebRTC/: a LibWebRTC package config providing the LibWebRTC::libwebrtc imported target
#    * <libdir>/pkgconfig/libwebrtc.pc on Linux
#
cmake_minimum_required(VERSION 3.16)

# C is only enabled so that GNUInstallDirs knows the target architecture (lib vs lib64)
project(libwebrtc LANGUAGES C)

if(NOT LIBWEBRTC_DIST_DIR OR NOT EXISTS "${LIBWEBRTC_DIST_DIR}/include/libwebrtc.h")
  message(FATAL_ERROR "LIBWEBRTC_DIST_DIR (${LIBWEBRTC_DIST_DIR}) is not an extracted libwebrtc distribution")
endif()

include(GNUInstallDirs)
include(CMakePackageConfigHelpers)

set(_LIBWEBRTC_INCLUDE_DEST "${CMAKE_INSTALL_INCLUDEDIR}/libwebrtc")
set(_LIBWEBRTC_CMAKE_DEST "${CMAKE_INSTALL_LIBDIR}/cmake/LibWebRTC")

# m150.7871.04 -> 150.7871.04 for the package version file
string(REGEX REPLACE "^m" "" _LIBWEBRTC_NUMERIC_VERSION "${LIBWEBRTC_VERSION}")

# the releases are built with libwebrtc_desktop_capture=true, which adds virtual methods to
# RTCPeerConnectionFactory under RTC_DESKTOP_DEVICE: consumers must see the same declarations or the vtables
# won't match
set(LIBWEBRTC_EXPORTED_DEFINES RTC_DESKTOP_DEVICE)

#
# headers, library and license
#
install(DIRECTORY "${LIBWEBRTC_DIST_DIR}/include/" DESTINATION "${_LIBWEBRTC_INCLUDE_DEST}")

if(WIN32)
  list(APPEND LIBWEBRTC_EXPORTED_DEFINES LIB_WEBRTC_API_DLL)
  install(FILES "${LIBWEBRTC_DIST_DIR}/lib/libwebrtc.dll" DESTINATION "${CMAKE_INSTALL_BINDIR}")
  install(FILES "${LIBWEBRTC_DIST_DIR}/lib/libwebrtc.dll.lib" DESTINATION "${CMAKE_INSTALL_LIBDIR}")
  set(_LIBWEBRTC_LOCATION_PROPS
"IMPORTED_LOCATION \"\${_LIBWEBRTC_PREFIX}/${CMAKE_INSTALL_BINDIR}/libwebrtc.dll\"
    IMPORTED_IMPLIB \"\${_LIBWEBRTC_PREFIX}/${CMAKE_INSTALL_LIBDIR}/libwebrtc.dll.lib\"")
else()
  install(FILES "${LIBWEBRTC_DIST_DIR}/lib/libwebrtc.so" DESTINATION "${CMAKE_INSTALL_LIBDIR}"
          PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
  set(_LIBWEBRTC_LOCATION_PROPS
"IMPORTED_LOCATION \"\${_LIBWEBRTC_PREFIX}/${CMAKE_INSTALL_LIBDIR}/libwebrtc.so\"
    IMPORTED_SONAME libwebrtc.so")
endif()

install(FILES "${LIBWEBRTC_DIST_DIR}/LICENSE" DESTINATION "${CMAKE_INSTALL_DOCDIR}")

#
# CMake package config and pkg-config file
#
file(RELATIVE_PATH LIBWEBRTC_PREFIX_FROM_CONFIG "${CMAKE_INSTALL_PREFIX}/${_LIBWEBRTC_CMAKE_DEST}" "${CMAKE_INSTALL_PREFIX}")

file(CONFIGURE OUTPUT "${CMAKE_BINARY_DIR}/LibWebRTCConfig.cmake" @ONLY CONTENT [=[
# LibWebRTC package config generated by accendino's libwebrtc/CMakeLists.txt
get_filename_component(_LIBWEBRTC_PREFIX "${CMAKE_CURRENT_LIST_DIR}/@LIBWEBRTC_PREFIX_FROM_CONFIG@" ABSOLUTE)

set(LibWebRTC_VERSION_STRING "@LIBWEBRTC_VERSION@")
set(LibWebRTC_INCLUDE_DIR "${_LIBWEBRTC_PREFIX}/@_LIBWEBRTC_INCLUDE_DEST@")

if(NOT TARGET LibWebRTC::libwebrtc)
  add_library(LibWebRTC::libwebrtc SHARED IMPORTED)
  set_target_properties(LibWebRTC::libwebrtc PROPERTIES
    @_LIBWEBRTC_LOCATION_PROPS@
    INTERFACE_INCLUDE_DIRECTORIES "${LibWebRTC_INCLUDE_DIR}"
    INTERFACE_COMPILE_DEFINITIONS "@LIBWEBRTC_EXPORTED_DEFINES@"
  )
endif()

unset(_LIBWEBRTC_PREFIX)
]=])

write_basic_package_version_file("${CMAKE_BINARY_DIR}/LibWebRTCConfigVersion.cmake"
  VERSION "${_LIBWEBRTC_NUMERIC_VERSION}" COMPATIBILITY SameMajorVersion)

install(FILES "${CMAKE_BINARY_DIR}/LibWebRTCConfig.cmake" "${CMAKE_BINARY_DIR}/LibWebRTCConfigVersion.cmake"
        DESTINATION "${_LIBWEBRTC_CMAKE_DEST}")

if(NOT WIN32)
  list(TRANSFORM LIBWEBRTC_EXPORTED_DEFINES PREPEND "-D" OUTPUT_VARIABLE _defines)
  list(JOIN _defines " " _cflags)

  file(WRITE "${CMAKE_BINARY_DIR}/libwebrtc.pc"
"prefix=${CMAKE_INSTALL_PREFIX}
includedir=\${prefix}/${_LIBWEBRTC_INCLUDE_DEST}
libdir=\${prefix}/${CMAKE_INSTALL_LIBDIR}

Name: libwebrtc
Description: C++ wrapper around Google's WebRTC (webrtc-sdk/libwebrtc binary distribution)
Version: ${_LIBWEBRTC_NUMERIC_VERSION}
Cflags: -I\${includedir} ${_cflags}
Libs: -L\${libdir} -lwebrtc
")
  install(FILES "${CMAKE_BINARY_DIR}/libwebrtc.pc" DESTINATION "${CMAKE_INSTALL_LIBDIR}/pkgconfig")
endif()
